Data description

I have kept track of over six hundred films and television shows that I have rated on IMDb over the past several years. Since these ratings reflect my own taste at the moment I watched each work, I wanted to study whether there are underlying patterns in the types of content I consistently enjoy and how my taste may have changed over time. The dataset includes my personal rating on a ten point scale, the date I rated the film, the IMDb rating and vote count, runtime, genre information, and more.

To enrich this data, I used the TMDb API to pull structured metadata for each title. This includes TMDb genres, release dates, popularity scores, cast and crew information, and runtime. TMDb provides a more consistent taxonomy of genres than IMDb, so I preferred it for my feature engineering. After merging the IMDb and TMDb information, the dataset became a compact record of what I have watched and how I felt about it, along with a set of detailed, structured attributes that enable machine learning analysis.

import os
import time
import requests

# CONFIG
TMDB_BEARER_TOKEN = os.environ.get("TMDB_BEARER_TOKEN")
if not TMDB_BEARER_TOKEN:
    raise ValueError("TMDB_BEARER_TOKEN is not set")

BASE_URL = "https://api.themoviedb.org/3"


def tmdb_get(path: str, params: dict | None = None, sleep: float = 0.1) -> dict:
    """
    Small helper for calling the TMDb API.
    Adds api_key automatically, handles JSON parsing, and sleeps briefly to respect rate limits.
    """
    if params is None:
        params = {}
    params = dict(params)

    url = f"{BASE_URL}{path}"
    resp = requests.get(
        url,
        params=params,
        headers={"Authorization": f"Bearer {TMDB_BEARER_TOKEN}"},
        timeout=30,
    )
    resp.raise_for_status()
    data = resp.json()

    # Be reasonably gentle with the API so it doesn't get stuck with rate limiting
    time.sleep(sleep)
    return data

def fetch_full_tmdb_record(imdb_id: str) -> dict | None:
    """
    Given an IMDb ID like 'tt0111161', fetch a rich TMDb record:
    basic movie or TV info (via /find), detailed metadata (via /movie or /tv), and credits (director and main cast).
    Returns a flat dict of features, or None if no match is found.
    """
    # 1. Find TMDb ID and basic info from IMDb ID
    find_data = tmdb_get(
        path=f"/find/{imdb_id}",
        params={"external_source": "imdb_id"},
        sleep=0.05,
    )

    if find_data.get("movie_results"):
        base = find_data["movie_results"][0]
        media_type = "movie"
    elif find_data.get("tv_results"):
        base = find_data["tv_results"][0]
        media_type = "tv"
    else:
        # No match (e.g., some episodes, very obscure titles)
        return None

    tmdb_id = base["id"]

    # 2. Detailed metadata (/movie/{id} or /tv/{id})
    details = tmdb_get(
        path=f"/{media_type}/{tmdb_id}",
        params={},
        sleep=0.05,
    )

    # 3. Credits: director and main cast (/credits)
    credits = tmdb_get(
        path=f"/{media_type}/{tmdb_id}/credits",
        params={},
        sleep=0.05,
    )

    # Extract director info
    director_name = director_id = None
    director_gender = None

    crew = credits.get("crew", [])
    directors = [c for c in crew if c.get("job") == "Director"]
    if directors:
        d0 = directors[0]
        director_name = d0.get("name")
        director_id = d0.get("id")
        director_gender = d0.get("gender")  # 1 = female, 2 = male, 0/None = unknown

    # Extract main cast (top-billed)
    main_cast_name = main_cast_id = None
    main_cast_gender = None

    cast = credits.get("cast", [])
    if cast:
        cast_sorted = sorted(cast, key=lambda a: a.get("order", 9999))
        main = cast_sorted[0]
        main_cast_name = main.get("name")
        main_cast_id = main.get("id")
        main_cast_gender = main.get("gender")  # 1 = female, 2 = male, 0/None = unknown

    # Genres (ids + names)
    genre_ids = [g["id"] for g in details.get("genres", [])]
    genre_names = [g["name"] for g in details.get("genres", [])]

    # Spoken languages and production countries
    spoken_languages = [lang.get("iso_639_1") for lang in details.get("spoken_languages", [])]
    production_countries = [c.get("iso_3166_1") for c in details.get("production_countries", [])]

    # Runtime
    runtime = details.get("runtime")
    if runtime is None and media_type == "tv":
        episode_run_time = details.get("episode_run_time") or []
        runtime = episode_run_time[0] if episode_run_time else None

    return {
        "tmdb_id": tmdb_id,
        "tmdb_media_type": media_type,
        "tmdb_title": details.get("title") or details.get("name"),
        "tmdb_original_title": details.get("original_title") or details.get("original_name"),
        "tmdb_release_date": details.get("release_date") or details.get("first_air_date"),
        "tmdb_original_language": details.get("original_language"),
        "tmdb_vote_average": details.get("vote_average"),
        "tmdb_vote_count": details.get("vote_count"),
        "tmdb_popularity": details.get("popularity"),

        "tmdb_genre_ids": genre_ids,
        "tmdb_genre_names": genre_names,
        "runtime": runtime,
        "status": details.get("status"),
        "adult": details.get("adult"),
        "overview": details.get("overview"),
        "spoken_languages": spoken_languages,
        "production_countries": production_countries,

        "director_id": director_id,
        "director_name": director_name,
        "director_gender": director_gender,

        "main_cast_id": main_cast_id,
        "main_cast_name": main_cast_name,
        "main_cast_gender": main_cast_gender,
    }

def parse_maybe_list(x):
    if isinstance(x, list):
        return x
    if isinstance(x, str):
        x = x.strip()
        if x.startswith("[") and x.endswith("]"):
            try:
                return ast.literal_eval(x)
            except Exception:
                return []
        elif x:  # non-empty plain string
            # for things like "Drama, Comedy" in imdb_genres if you want later
            return [s.strip() for s in x.split(",")]
    return []

def embed_batch(paths: list[str], batch_size: int = 32) -> np.ndarray:
    embs = []
    for start in range(0, len(paths), batch_size):
        batch_paths = paths[start:start + batch_size]
        imgs = []
        for p in batch_paths:
            img = Image.open(p).convert("RGB")
            imgs.append(preprocess(img))
        x = torch.stack(imgs, dim=0).to(device)
        with torch.no_grad():
            out = model(x).detach().cpu().numpy()
        embs.append(out)
    return np.concatenate(embs, axis=0) if embs else np.zeros((0, 2048), dtype=np.float32)

Multimodal ICA (metadata + posters + text)

Up to now, ICA has only seen the structured metadata feature space. Since I now have three representations per title (metadata, poster CNN embeddings, and text embeddings), I can also run ICA on a multimodal feature vector.

I compare two versions:

  1. Naive concat: [Zmeta,Zposter,Ztext][Z_{meta}, Z_{poster}, Z_{text}] after PCA-per-block + standardization.
  2. Residualized: [Zmeta,Rposter,Rtext][Z_{meta}, R_{poster}, R_{text}] where RR removes the linear part that is predictable from metadata (so the remaining signal is closer to “what posters/text add beyond metadata”).

Interpretation note: ICA components are still only defined up to sign and permutation, so I treat them as axes rather than “IC1 is always the same concept”.

# How much each modality contributes to each IC (via unmixing weights)
def block_norms(ica):
    rows = []
    for j in range(ica.components_.shape[0]):
        w = ica.components_[j]
        rows.append({
            "IC": j+1,
            "meta_norm": float(np.linalg.norm(w[:k_meta])),
            "poster_norm": float(np.linalg.norm(w[k_meta:k_meta+k_post])),
            "text_norm": float(np.linalg.norm(w[k_meta+k_post:])),
        })
    out = pd.DataFrame(rows)
    out["total"] = out[["meta_norm", "poster_norm", "text_norm"]].sum(axis=1)
    for col in ["meta_norm", "poster_norm", "text_norm"]:
        out[col.replace("_norm", "_share")] = out[col] / out["total"].replace(0, np.nan)
    return out.drop(columns=["total"])
